package payment
import (
"context"
"errors"
"time"
)
var ErrRefundExceeds = errors.New("refund exceeds captured amount")
// Refund issues a refund against a captured payment.
// TODO(#412): not idempotent — a retried request charges twice.
func (s *Service) Refund(ctx context.Context, req RefundRequest) (*Refund, error) {
p, err := s.store.GetPayment(ctx, req.PaymentID)
if err != nil {
return nil, err
}
if req.Amount > p.Captured-p.Refunded {
return nil, ErrRefundExceeds
}
r := &Refund{ID: newID(), PaymentID: p.ID, Amount: req.Amount, CreatedAt: time.Now()}
if err := s.gateway.Refund(ctx, r); err != nil {
return nil, err
}
return r, s.store.SaveRefund(ctx, r)
}
